-- primary
-- mysqld.cnf


[mysqld]
server_id=1

log_bin=mysql-bin
binlog_format=ROW

gtid_mode=ON
enforce_gtid_consistency=ON

log_replica_updates=ON


-- restart mysql

CREATE USER 'repl'@'%' IDENTIFIED BY 'StrongPassword';
GRANT REPLICATION SLAVE ON *.* TO 'repl'@'%';
FLUSH PRIVILEGES;

-----------------------------------------------------------------

-- replica 1

[mysqld]
server_id=2

relay_log=relay-bin

gtid_mode=ON
enforce_gtid_consistency=ON

log_bin=mysql-bin
log_replica_updates=ON

read_only=ON
super_read_only=ON

-----------------------------------------------------------------


-- replica 2

[mysqld]
server_id=3

relay_log=relay-bin

gtid_mode=ON
enforce_gtid_consistency=ON

log_bin=mysql-bin
log_replica_updates=ON

read_only=ON
super_read_only=ON

-----------------------------------------------------------------

-- both replicas

CHANGE REPLICATION SOURCE TO
    SOURCE_HOST='10.10.1.31',
    SOURCE_PORT=3309,
    SOURCE_USER='repl',
    SOURCE_PASSWORD='StrongPassword',
    SOURCE_AUTO_POSITION=1;

START REPLICA;

-----------------------------------------------------------------

-- check both replicas

SHOW REPLICA STATUS\G
