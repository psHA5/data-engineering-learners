--server
docker volume create pmm-data
sudo docker run -d \
  -p 80:80 -p 443:443 \
  -v pmm-data:/srv \
  --name pmm-server \
  --restart always \
  percona/pmm-server:2

--client

sudo docker run -d \
  --name pmm-client \
  --network mysql_mysqlnet \
  percona/pmm-client:3


-- mysql monitored servers

CREATE USER 'pmm'@'%' IDENTIFIED BY 'password';

GRANT SELECT, PROCESS, REPLICATION CLIENT ON *.* TO 'pmm'@'%';

GRANT SELECT ON performance_schema.* TO 'pmm'@'%';

FLUSH PRIVILEGES;
