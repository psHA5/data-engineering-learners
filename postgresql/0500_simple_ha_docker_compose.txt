services:

  postgres-primary:
    image: postgres:18
    container_name: postgres-primary
    hostname: postgres-primary
    restart: unless-stopped
    environment:
      POSTGRES_DB: postgres
      POSTGRES_USER: postgres
      POSTGRES_PASSWORD: postgres
    command:
      - postgres
      - -c
      - wal_level=replica
      - -c
      - max_wal_senders=10
      - -c
      - max_replication_slots=10
      - -c
      - hot_standby=on
    ports:
      - "5432:5432"
    volumes:
      - primary-data:/var/lib/postgresql
      - ./primary/init:/docker-entrypoint-initdb.d
    networks:
      - postgres-net

  postgres-replica:
    image: postgres:18
    container_name: postgres-replica
    hostname: postgres-replica
    restart: unless-stopped
    environment:
      POSTGRES_PASSWORD: postgres
    ports:
      - "5433:5432"
    volumes:
      - replica-data:/var/lib/postgresql
    depends_on:
      - postgres-primary
    networks:
      - postgres-net

  pgbouncer:
    image: edoburu/pgbouncer:latest
    container_name: pgbouncer
    restart: unless-stopped
    ports:
      - "6432:6432"
    environment:
      POSTGRESQL_HOST: postgres-primary
      POSTGRESQL_PORT: 5432
      POSTGRESQL_USERNAME: postgres
      POSTGRESQL_PASSWORD: postgres
      PGBOUNCER_DATABASE: postgres
      PGBOUNCER_POOL_MODE: transaction
      PGBOUNCER_MAX_CLIENT_CONN: 1000
      PGBOUNCER_DEFAULT_POOL_SIZE: 25
    depends_on:
      - postgres-primary
    networks:
      - postgres-net

volumes:
  primary-data:
  replica-data:

networks:
  postgres-net:
    driver: bridge
